#include <stdio.h> #include <unistd.h> #include <sys/wait.h> int main() {
pid_t pid=fork(); if(pid<0) {
perror("fork"); return 1;
}
if(pid==0) {
printf("Child PID = %d\n",getpid()); printf("Initial Parent PID = %d\n",getppid()); sleep(5);
printf("After parent exits, Parent PID = %d\n",getppid()); printf("Child is now an orphan process.\n");
} else {
printf("Parent PID = %d\n",getpid()); printf("Parent is terminating now.\n");
}
return 0;
}

#include <stdio.h> #include <stdlib.h> #include <string.h> #include <unistd.h> #include <sys/wait.h> int main() {
char line[300]; while(1) {
char *args[20]; int n=0; printf("$ ");
fgets(line,sizeof(line),stdin); line[strcspn(line,"\n")]='\0'; if(strcmp(line,"exit")==0) break; char *token=strtok(line," "); while(token!=NULL) {
args[n++]=token; token=strtok(NULL," ");
}
args[n]=NULL;
if(n==4 && strcmp(args[0],"search")==0) { FILE *fp=fopen(args[2],"r");
char word[100];
int found=0, total=0; if(fp==NULL) {
printf("File cannot be opened.\n"); continue;
}
while(fscanf(fp,"%99s",word)==1) { if(strcmp(word,args[3])==0) {
total++; if(strcmp(args[1],"f")==0) {
printf("First occurrence found: %s\n",word); found=1;
break;
}
}
}
fclose(fp);
if(strcmp(args[1],"a")==0) printf("Total occurrences = %d\n",total); else if(strcmp(args[1],"c")==0) printf("Count = %d\n",total);
else if(strcmp(args[1],"f")==0 && !found) printf("Pattern not found.\n");
} else {
pid_t pid=fork(); if(pid==0) {
execvp(args[0],args); perror("Command failed"); exit(1);
}
wait(NULL);
}
}
return 0;
