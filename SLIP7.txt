#include <stdio.h> #define P 5
#define R 4 int main() {
int allocation[P][R] = {
{2, 0, 0, 1},
{3, 1, 2, 1},
{2, 1, 0, 3},
{1, 3, 1, 2},
{1, 4, 3, 2}
};
int maximum[P][R] = {
{4, 2, 1, 2},
{5, 2, 5, 2},
{2, 3, 1, 6},
{1, 4, 2, 4},
{3, 6, 6, 5}
};
int available[R] = {3, 3, 2, 1}; int need[P][R];
int i,j,choice; for(i=0;i<P;i++)
for(j=0;j<R;j++)
need[i][j]=maximum[i][j]-allocation[i][j]; while(1) {
printf("\n1. Accept Available\n"); printf("2. Display Allocation and Max\n"); printf("3. Display Need\n");
printf("4. Display Available\n"); printf("5. Check Safe Sequence\n"); printf("6. Exit\n");
printf("Enter choice: "); scanf("%d",&choice); if(choice==1) {
printf("Enter %d Available values: ",R); for(j=0;j<R;j++) scanf("%d",&available[j]);
}
else if(choice==2) { printf("Allocation and Max:\n"); for(i=0;i<P;i++) {
printf("P%d ",i);
for(j=0;j<R;j++) printf("%d ",allocation[i][j]); printf(" | ");
for(j=0;j<R;j++) printf("%d ",maximum[i][j]); printf("\n");
}
}
else if(choice==3) { printf("Need Matrix:\n"); for(i=0;i<P;i++) {
printf("P%d: ",i);
for(j=0;j<R;j++) printf("%d ",need[i][j]); printf("\n");
}
}
else if(choice==4) { printf("Available: ");
for(j=0;j<R;j++) printf("%d ",available[j]); printf("\n");
}
else if(choice==5) {
int work[R], finish[P]={0}, safe[P], count=0, found; for(j=0;j<R;j++) work[j]=available[j]; while(count<P) {
found=0; for(i=0;i<P;i++) {
if(!finish[i]) { int possible=1;
for(j=0;j<R;j++) if(need[i][j]>work[j]) possible=0;
if(possible) {
for(j=0;j<R;j++) work[j]+=allocation[i][j]; finish[i]=1;
safe[count++]=i; found=1;
}
}
}
if(!found) break;
}
if(count==P) {
printf("System is SAFE.\nSafe sequence: "); for(i=0;i<P;i++) printf("P%d ",safe[i]); printf("\n");
} else {
printf("System is NOT SAFE.\n");
}
}

else if(choice==6) break;
else printf("Invalid choice.\n");
}
return 0;
}
