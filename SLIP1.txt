#include<stdio.h>
#define P 5
#define R 3
int main()
{
int allocation[P][R] = {
{2, 3, 2},
{4, 0, 0},
{5, 0, 4},
{4, 3, 3},
{2, 2, 4}
};
int maximum[P][R] = {
{9, 7, 5},
{5, 2, 2},
{1, 0, 4},
{4, 4, 4},
{6, 5, 5}
};
int available[R] = {3, 3, 2}; 
int need[P][R];
int i,j,choice;
 for(i=0;i<P;i++)
for(j=0;j<R;j++)
need[i][j]=maximum[i][j]-allocation[i][j]; 
while(1)
{
printf("\n1. Accept Available\n");
printf("2. Display Allocation and Max\n"); 
printf("3. Display Need\n");
printf("4. Display Available\n"); 
printf("5. Check Safe Sequence\n"); 
printf("6. Exit\n");
printf("Enter choice: "); 
scanf("%d",&choice); 
if(choice==1) 
{
printf("Enter %d Available values: ",R); 
for(j=0;j<R;j++) 
scanf("%d",&available[j]);
}
else if(choice==2) 
{ 
printf("Allocation and Max:\n");
for(i=0;i<P;i++) 
{
printf("P%d ",i);
for(j=0;j<R;j++)
 printf("%d ",allocation[i][j]); 
printf(" | ");
for(j=0;j<R;j++) 
printf("%d ",maximum[i][j]);
 printf("\n");
}
}
else if(choice==3)
 {
 printf("Need Matrix:\n");
 for(i=0;i<P;i++)
 {
printf("P%d: ",i);
for(j=0;j<R;j++)
 printf("%d ",need[i][j]);
 printf("\n");
}
}
else if(choice==4) 
{
 printf("Available: ");
for(j=0;j<R;j++)
 printf("%d ",available[j]);
 printf("\n");
}
else if(choice==5)
 {
int work[R], finish[P]={0}, safe[P], count=0, found; 
for(j=0;j<R;j++)
 work[j]=available[j];
 while(count<P)
 {
found=0;
 for(i=0;i<P;i++)
 {
if(!finish[i])
 { 
int possible=1;
for(j=0;j<R;j++) 
if(need[i][j]>work[j])
 possible=0;
if(possible) 
{
for(j=0;j<R;j++)
 work[j]+=allocation[i][j];
 finish[i]=1;
safe[count++]=i; 
found=1;
}
}
}
if(!found)
 break;
}
if(count==P)
 {
printf("System is SAFE.\nSafe sequence: ");
 for(i=0;i<P;i++)
 printf("P%d ",safe[i]);
 printf("\n");
} 
else {
printf("System is NOT SAFE.\n");
}
}
else if(choice==6)
 break;
else
 printf("Invalid choice.\n");
}
return 0;
}

#include <stdio.h>
 #include <stdlib.h>
 #include <string.h>
 #include <unistd.h>
 #include <sys/wait.h> 
int main()
 {
char line[200];
 while(1)
 {
char *args[20];
 int count=0, i; 
printf("$ ");
fgets(line,sizeof(line),stdin);
 line[strcspn(line,"\n")]='\0';
 if(strcmp(line,"exit")==0)
 break;
 char *token=strtok(line," ");
 while(token!=NULL) 
{
args[count++]=token;
 token=strtok(NULL," ");
}
args[count]=NULL;
if(count>=3 && strcmp(args[0],"count")==0)
 {
 FILE *fp=fopen(args[2],"r");
int ch, chars=0, words=0, lines=0, inword=0;
 if(fp==NULL)
 {
printf("File cannot be opened.\n");
 continue;
}
while((ch=fgetc(fp))!=EOF)
 {
 chars++;
if(ch=='\n')
 lines++;
if(ch==' ' || ch=='\n' || ch=='\t')
 inword=0; 
else if(!inword)
 {
 words++;
 inword=1;
 }
}
fclose(fp);
if(strcmp(args[1],"c")==0)
 printf("Characters = %d\n",chars);
 else if(strcmp(args[1],"w")==0)
 printf("Words = %d\n",words);
 else if(strcmp(args[1],"l")==0)
 printf("Lines = %d\n",lines);
 else 
printf("Use: count c/w/l filename\n");
}
 else {
pid_t pid=fork(); 
if(pid==0)
 {
execvp(args[0],args);
 perror("Command failed");
 exit(1);
}
wait(NULL);
}
}
return 0;
}
