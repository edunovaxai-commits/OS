int ref[] = {3, 5, 7, 2, 5, 1, 2, 3, 1, 3, 5, 3, 1, 6, 2};
int n=15, f, i,j,k,time=0,fault=0,found,pos,min; printf("Enter number of frames: "); scanf("%d",&f);
int frame[f], counter[f];
for(i=0;i<f;i++) { frame[i]=-1; counter[i]=0; } for(i=0;i<n;i++) {
time++; found=0;
for(j=0;j<f;j++)
if(frame[j]==ref[i]) { found=1; counter[j]=time; } if(!found) {
pos=-1;
for(j=0;j<f;j++) if(frame[j]==-1) {pos=j;break;} if(pos==-1) {
min=counter[0]; pos=0;
for(j=1;j<f;j++) if(counter[j]<min) {min=counter[j];pos=j;}
}
frame[pos]=ref[i]; counter[pos]=time; fault++;
}
printf("%d : ",ref[i]);
for(k=0;k<f;k++) printf("%d ",frame[k]); if(!found) printf(" Page Fault"); printf("\n");
}
printf("Total page faults = %d\n",fault); return 0;
}

#include <stdio.h> #include <stdlib.h> #include <string.h> #include <unistd.h> #include <sys/wait.h> int main() {
char line[200]; while(1) {
char *args[20]; int count=0, i; printf("$ ");
fgets(line,sizeof(line),stdin); line[strcspn(line,"\n")]='\0'; if(strcmp(line,"exit")==0) break; char *token=strtok(line," "); while(token!=NULL) {
args[count++]=token; token=strtok(NULL," ");
}
args[count]=NULL;
if(count>=3 && strcmp(args[0],"count")==0) { FILE *fp=fopen(args[2],"r");
int ch, chars=0, words=0, lines=0, inword=0; if(fp==NULL) {
printf("File cannot be opened.\n"); continue;
}
while((ch=fgetc(fp))!=EOF) { chars++;
if(ch=='\n') lines++;
if(ch==' ' || ch=='\n' || ch=='\t') inword=0; else if(!inword) { words++; inword=1; }
}
fclose(fp);
if(strcmp(args[1],"c")==0) printf("Characters = %d\n",chars); else if(strcmp(args[1],"w")==0) printf("Words = %d\n",words); else if(strcmp(args[1],"l")==0) printf("Lines = %d\n",lines); else printf("Use: count c/w/l filename\n");
} else {
pid_t pid=fork();

if(pid==0) {
execvp(args[0],args);
perror("Command failed"); exit(1);
}
wait(NULL);
}
}
return 0;
}
