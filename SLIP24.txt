#include <stdio.h>
#include <stdlib.h>
int main(int argc,char *argv[]) { int n=argc-2;
int key=atoi(argv[argc-1]); int low=0,high=n-1,mid;
int a[50],i;
for(i=0;i<n;i++) a[i]=atoi(argv[i+1]); while(low<=high) {
mid=(low+high)/2; if(a[mid]==key) {
printf("Item %d found at position %d.\n",key,mid+1); return 0;
}
if(a[mid]<key) low=mid+1; else high=mid-1;
}
printf("Item %d not found.\n",key); return 0;
}
