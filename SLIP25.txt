#include <stdio.h> int main() {
int n,i,time=0,done=0,q;
int at[20],bt[20],rem[20],ct[20],tat[20],wt[20]; int queue[100],front=0,rear=0,inqueue[20]={0}; float avgwt=0,avgtat=0;
printf("Enter number of processes: "); scanf("%d",&n);
for(i=0;i<n;i++) {
printf("Enter AT and first CPU burst for P%d: ",i+1); scanf("%d%d",&at[i],&bt[i]);
rem[i]=bt[i];
}
printf("Enter time quantum: "); scanf("%d",&q);
printf("\nGantt Chart:\n"); while(done<n) {
for(i=0;i<n;i++)
if(at[i]<=time && rem[i]>0 && !inqueue[i]) { queue[rear++]=i;
inqueue[i]=1;
}
if(front==rear) { time++; continue;
}
int p=queue[front++];
int run=(rem[p]<q)?rem[p]:q; printf("|P%d ",p+1);
rem[p]-=run; time+=run; for(i=0;i<n;i++)
if(at[i]<=time && rem[i]>0 && !inqueue[i]) { queue[rear++]=i;
inqueue[i]=1;
}
if(rem[p]>0)
queue[rear++]=p; else {
ct[p]=time; done++;
}
}
printf("|\n"); for(i=0;i<n;i++) {
tat[i]=ct[i]-at[i];
wt[i]=tat[i]-bt[i]; avgwt+=wt[i]; avgtat+=tat[i];
}
printf("Process\tAT\tBT\tCT\tTAT\tWT\n"); for(i=0;i<n;i++)
printf("P%d\t%d\t%d\t%d\t%d\t%d\n",i+1,at[i],bt[i],ct[i],tat[i],wt[i]); printf("Average WT = %.2f\n",avgwt/n);
printf("Average TAT = %.2f\n",avgtat/n); return 0;
}

for(j=0;j<f;j++)
if(frame[j]==ref[i]) {found=1;freq[j]++;age[j]=time;}

if(!found) {
pos=-1;
